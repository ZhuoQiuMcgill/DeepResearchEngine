/**
 * A cheap check of checklist coverage (design section 33): the judge's coverage instruction, word for
 * word, answered by Claude Haiku, for the recorded reports of both sides and for rewritten reports.
 * The recorded reports were scored by the Opus judge, so the first two rows show how far Haiku's
 * scores agree with it. A screen, not a judgment.
 *
 *   tsx local/ebd-heldout/cover-screen.ts <name ...>      name: baseline | ebd-one | a folder rewrite-final-<name>
 */
import "dotenv/config";
import { readFileSync, readdirSync, writeFileSync } from "node:fs";
import { join } from "node:path";
import { HumanMessage, SystemMessage } from "@langchain/core/messages";
import { createChatModel, getProfile } from "../../src/config/models.js";

const OUT = "local/ebd-heldout";
const names = process.argv.slice(2);
type Q = { id: string; question: string; checklist: string[] };
type Summary = { dir: string; strategy: { label: string }; tags: { task: string } };
const QUESTIONS = JSON.parse(readFileSync(join(OUT, "final-sealed-questions.json"), "utf8")) as Q[];
const folder = readdirSync(OUT).filter((d) => d.startsWith("final-sealed-2")).sort().at(-1)!;
const runs = JSON.parse(readFileSync(join(OUT, folder, "summaries.json"), "utf8")) as Summary[];
const report = (task: string, name: string) =>
  name === "baseline" || name === "ebd-one" ? readFileSync(join(runs.find((s) => s.tags.task === task && s.strategy.label === name)!.dir, "report.md"), "utf8") : readFileSync(join(OUT, `rewrite-final-${name}`, `${task}.md`), "utf8");
const COVERAGE_SYSTEM = /const COVERAGE_SYSTEM = `([\s\S]*?)`;/.exec(readFileSync(join(OUT, "judge.ts"), "utf8"))![1]!;
const model = createChatModel(getProfile("haiku"));
const usage = { input: 0, output: 0 };

async function cover(q: Q, name: string): Promise<{ name: string; task: string; scores: number[] }> {
  const human = `Question:\n${q.question}\n\nChecklist:\n${q.checklist.map((c, i) => `${i + 1}. ${c}`).join("\n")}\n\n<report>\n${report(q.id, name)}\n</report>`;
  for (let attempt = 0; attempt < 3; attempt++) {
    const msg = await model.invoke([new SystemMessage(COVERAGE_SYSTEM), new HumanMessage(human)]);
    const u = (msg as { usage_metadata?: { input_tokens?: number; output_tokens?: number } }).usage_metadata;
    usage.input += u?.input_tokens ?? 0;
    usage.output += u?.output_tokens ?? 0;
    const raw = typeof msg.content === "string" ? msg.content : (msg.content as { type: string; text?: string }[]).filter((b) => b.type === "text").map((b) => b.text ?? "").join("");
    try {
      const items = (JSON.parse(raw.match(/\{[\s\S]*\}/)![0]) as { items: { n: number; score: number }[] }).items;
      if (items.length === q.checklist.length) return { name, task: q.id, scores: items.map((it) => Number(it.score)) };
    } catch { /* asked again */ }
  }
  return { name, task: q.id, scores: [] };
}

const jobs = names.flatMap((name) => QUESTIONS.map((q) => () => cover(q, name)));
const results: Awaited<ReturnType<typeof cover>>[] = [];
let next = 0;
await Promise.all(Array.from({ length: 6 }, async () => {
  while (next < jobs.length) results.push(await jobs[next++]!());
}));
writeFileSync(join(OUT, "cover-screen.json"), JSON.stringify({ usage, results }, null, 1));
console.log(`${results.length} coverage checks by Haiku; tokens in ${usage.input}, out ${usage.output}; about $${((usage.input * 0.1 + usage.output * 0.5) / 1e6).toFixed(2)}`);
